create database INSURANCE_DATABASE;
use INSURANCE_DATABASE;
CREATE TABLE person (
    driver_id INT PRIMARY KEY, driver_name VARCHAR(100), address VARCHAR(300)
);
CREATE TABLE car (reg_number int PRIMARY KEY, model varchar(50), reg_year int(100) );
CREATE TABLE owns(driver_id INT, FOREIGN KEY (driver_id) REFERENCES person(driver_id), reg_number INT, FOREIGN KEY (reg_number) REFERENCES car(reg_number)
    );
CREATE TABLE accident ( report_num INT PRIMARY KEY, accident_date varchar(10), location varchar(100));
CREATE TABLE participated ( driver_id INT, FOREIGN KEY (driver_id) REFERENCES person(driver_id), reg_number INT, FOREIGN KEY (reg_number) REFERENCES car(reg_number), report_num INT, FOREIGN KEY(report_num) REFERENCES accident(report_num), damage_amount INT);
